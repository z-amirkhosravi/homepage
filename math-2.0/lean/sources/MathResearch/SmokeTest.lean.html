theorem square_expansion (x y : ℚ) :
    (x + y) ^ 2 = x ^ 2 + 2 * x * y + y ^ 2 := by
  ring

theorem sum_add {ι : Type*} (s : Finset ι) (f g : ι → ℚ) :
    (∑ i ∈ s, (f i + g i)) = (∑ i ∈ s, f i) + ∑ i ∈ s, g i := by
  exact Finset.sum_add_distrib
